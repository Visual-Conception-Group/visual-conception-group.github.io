{%- comment -%} Word cloud of the most frequent words in paper titles (home page hero). Usage: {% include word-cloud.html pubs=published %} Settings (number of words, stop words, merged spellings) are in _data/wordcloud.yml. The words are counted here, at build time (once per title). Each word links to /publications/?word=,, which publications.js matches against title words with the same rules, so the counts and the results agree. assets/js/word-cloud.js packs them into a cloud; without JavaScript they show as a simple wrapped list. {%- endcomment -%} {%- assign wc = site.data.wordcloud -%} {%- assign stop = wc.stopwords -%} {%- comment -%} Each word counts once per title, so "in N paper titles" is exact. {%- endcomment -%} {%- assign words = "" | split: "" -%} {%- for p in include.pubs -%} {%- assign t = p.title | downcase | replace: ":", " " | replace: ",", " " | replace: "(", " " | replace: ")", " " | replace: "?", " " | replace: "#", " " | replace: ".", " " | replace: '"', " " | replace: "'", " " -%} {%- assign title_words = "" | split: "" -%} {%- assign tokens = t | split: " " -%} {%- for token in tokens -%} {%- assign w = token | strip -%} {%- if w.size < 3 or stop contains w -%}{%- continue -%}{%- endif -%} {%- if wc.merge[w] -%}{%- assign w = wc.merge[w] -%}{%- endif -%} {%- assign title_words = title_words | push: w -%} {%- endfor -%} {%- assign title_words = title_words | uniq -%} {%- for w in title_words -%}{%- assign words = words | push: w -%}{%- endfor -%} {%- endfor -%} {%- assign counted = words | group_by_exp: "w", "w" | sort: "size" | reverse -%} {%- assign top = counted.first.size | default: 1 -%} {%- assign shown = 0 %}
